    // ── Tabla ────────────────────────────────────────────────────────────────
    function actualizarTabla() {
        const tbody = document.getElementById('metricsTableBody');
        tbody.innerHTML = '';

        const sizes = Object.keys(metricsData).sort((a,b) => parseInt(a)-parseInt(b));

        sizes.forEach(size => {
            const m = metricsData[size];

            function t(algo) {
                return (m[algo]?.time_ms_mean !== undefined)
                    ? m[algo].time_ms_mean.toFixed(2)
                    : (m.time_ms || 0).toFixed(2);
            }
            function mem(algo) {
                return (m[algo]?.mem_bytes_peak_mean !== undefined)
                    ? (m[algo].mem_bytes_peak_mean / (1024*1024)).toFixed(3)
                    : (m.memory_mb || 0).toFixed(3);
            }
            function cls(v) {
                const n = parseFloat(v);
                return n > 1000 ? 'value-error' : n > 100 ? 'value-warning' : 'value-good';
            }

            const tr = document.createElement('tr');
            tr.innerHTML = `
                <td><strong>N = ${size}</strong></td>
                <td class="${cls(t('dfs'))}">${t('dfs')}</td>
                <td class="${cls(t('ldfs'))}">${t('ldfs')}</td>
                <td class="${cls(t('idfs'))}">${t('idfs')}</td>
                <td class="${parseFloat(mem('dfs')) > 100 ? 'value-warning' : 'value-good'}">${mem('dfs')}</td>
            `;
            tbody.appendChild(tr);
        });

        document.getElementById('alertasCount').textContent = alertasActuales.length;
    }
